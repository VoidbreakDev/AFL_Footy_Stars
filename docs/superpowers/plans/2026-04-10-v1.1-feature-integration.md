```tsx
import React from 'react';
import { useGameContext } from '../context/GameContext';

const SAVE_KEY = (slot: number) => `footyLegendSave_slot${slot}`;

interface SlotSnapshot {
  name: string;
  club: string;
  age: number;
  level: number;
  tier: string;
}

const getSlotSnapshot = (slot: number): SlotSnapshot | null => {
  try {
    const raw = localStorage.getItem(SAVE_KEY(slot));
    if (!raw) return null;
    const data = JSON.parse(raw);
    const p = data.player;
    return {
      name: p.name,
      club: p.contract?.clubName ?? 'Unknown',
      age: p.age,
      level: p.level,
      tier: p.contract?.tier ?? 'LOCAL',
    };
  } catch {
    return null;
  }
};

const SlotSelect: React.FC = () => {
  const { setCurrentSlot, loadGame, setView } = useGameContext();

  const handleSelectSlot = (slot: number) => {
    setCurrentSlot(slot);
    const loaded = loadGame();
    if (!loaded) {
      setView('ONBOARDING');
    } else {
      setView('DASHBOARD');
    }
  };

  return (
    <div className="min-h-screen bg-gradient-to-br from-green-900 to-green-700 flex flex-col items-center justify-center p-6">
      <h1 className="text-3xl font-black text-white mb-2">AFL Footy Stars</h1>
      <p className="text-green-200 mb-8">Select a career to continue or start a new one.</p>
      <div className="flex flex-col gap-4 w-full max-w-sm">
        {[0, 1, 2].map(slot => {
          const snap = getSlotSnapshot(slot);
          return (
            <button
              key={slot}
              onClick={() => handleSelectSlot(slot)}
              className="bg-white/10 hover:bg-white/20 border border-white/20 rounded-xl p-4 text-left transition"
            >
              {snap ? (
                <>
                  <div className="text-white font-bold">{snap.name}</div>
                  <div className="text-green-200 text-sm">{snap.club} · Age {snap.age} · Lvl {snap.level} · {snap.tier}</div>
                  <div className="text-green-400 text-xs mt-1">Career Slot {slot + 1}</div>
                </>
              ) : (
                <div className="text-white/60 flex items-center gap-2">
                  <span className="text-2xl">+</span>
                  <span>New Career (Slot {slot + 1})</span>
                </div>
              )}
            </button>
          );
        })}
      </div>
    </div>
  );
};

export default SlotSelect;
```

```tsx
import React from 'react';
import { useGameContext } from '../context/GameContext';

interface TipCardProps {
  tipKey: string;
  title: string;
  body: string;
}

const TipCard: React.FC<TipCardProps> = ({ tipKey, title, body }) => {
  const { player, setPlayer, saveGame } = useGameContext();

  if (!player) return null;
  // For existing saves (seenTips undefined), default to all tips shown (don't display).
  if (player.seenTips !== undefined && player.seenTips[tipKey]) return null;
  // New players (seenTips is undefined → show tips)
  // Returning players (seenTips exists, key missing → also show until dismissed)

  const dismiss = () => {
    setPlayer(prev => {
      if (!prev) return null;
      return { ...prev, seenTips: { ...(prev.seenTips ?? {}), [tipKey]: true } };
    });
    saveGame();
  };

  return (
    <div className="bg-blue-900/80 border border-blue-400/30 rounded-xl p-4 mb-4 flex items-start gap-3">
      <span className="text-blue-300 text-xl shrink-0">💡</span>
      <div className="flex-1 min-w-0">
        <div className="text-blue-200 font-bold text-sm">{title}</div>
        <div className="text-blue-100/80 text-xs mt-1 leading-relaxed">{body}</div>
      </div>
      <button
        onClick={dismiss}
        className="text-blue-300 hover:text-white text-xl font-bold shrink-0 leading-none"
        aria-label="Dismiss tip"
      >
        ×
      </button>
    </div>
  );
};

export default TipCard;
```

```typescript
// Position-specific event branching
switch (player.position) {
  case Position.FORWARD: {
    if (Math.random() < 0.35) {
```

```typescript
import { PlayerProfile } from '../types';

export const calculateLegacyScore = (player: PlayerProfile): number => {
  let score = 0;

  // Awards — iterate careerHistory
  const allAwards = player.careerHistory?.flatMap(h => (h as any).awards ?? []) ?? [];
  allAwards.forEach((award: any) => {
    switch (award.type) {
      case 'BROWNLOW_MEDAL': score += 100; break;
      case 'COLEMAN_MEDAL':  score += 80;  break;
      case 'ALL_AUSTRALIAN': score += 50;  break;
      case 'BEST_AND_FAIREST': score += 30; break;
      default: score += 20; break;
    }
  });

  // Premierships
  const premierships = player.careerHistory?.filter((h: any) => h.grandFinalWon).length ?? 0;
  score += premierships * 150;

  // Career stats
  score += Math.min(player.careerStats?.matches ?? 0, 300);
  score += Math.floor(Math.min(player.careerStats?.goals ?? 0, 200) * 0.5);

  // Milestones achieved
  const milestonesAchieved = player.milestones?.filter((m: any) => m.achieved).length ?? 0;
  score += milestonesAchieved * 10;

  // Seasons in AFL/NATIONAL tier
  const aflSeasons = player.careerHistory?.filter((h: any) => h.tier === 'NATIONAL').length ?? 0;
  score += aflSeasons * 20;

  return Math.round(score);
};

export const getLegacyTier = (score: number): string => {
  if (score >= 900) return 'All-Time Legend';
  if (score >= 700) return 'Superstar';
  if (score >= 500) return 'AFL Star';
  if (score >= 300) return 'State Great';
  if (score >= 100) return 'Club Legend';
  return 'Local Hero';
};
```

```typescript
  if (player.age < 30) return player;

      speed: Math.max(20, player.attributes.speed - speedDecline),
      stamina: Math.max(20, player.attributes.stamina - staminaDecline),
      kicking: Math.min(player.potential, player.attributes.kicking + (player.age >= 30 ? 1 : 0)),
```

```typescript
const completeDrill = (drill: DrillType, score: number) => {
  sessionStorage.setItem(drillSessionKey, '1');
  setDrillDone(true);
  setActiveDrill(false);

  let attrKey: keyof PlayerProfile['attributes'] | null = null;
  let attrBonus = 0;
  let xpBonus = 0;
  let msg = '';

  if (drill === 'kick') {
    attrKey = 'kicking';
    if (score >= 5) { attrBonus = 2; xpBonus = 300; msg = '+2 Kicking, +300 XP'; }
    else if (score >= 3) { attrBonus = 1; xpBonus = 150; msg = '+1 Kicking, +150 XP'; }
    else { xpBonus = 50; msg = '+50 XP'; }
  } else if (drill === 'speed') {
    attrKey = 'speed';
    // score is reaction time in ms (lower is better)
    if (score < 500) { attrBonus = 2; xpBonus = 200; msg = '+2 Speed, +200 XP'; }
    else if (score < 1000) { attrBonus = 1; xpBonus = 100; msg = '+1 Speed, +100 XP'; }
    else { xpBonus = 50; msg = '+50 XP'; }
  } else {
    attrKey = 'tackling';
    // score is click count
    if (score >= 20) { attrBonus = 2; xpBonus = 250; msg = '+2 Tackling, +250 XP'; }
    else if (score >= 10) { attrBonus = 1; xpBonus = 150; msg = '+1 Tackling, +150 XP'; }
    else { xpBonus = 50; msg = '+50 XP'; }
  }

  setDrillResult(`Drill Complete — ${msg}`);

  // Apply reward via setPlayer — cap at potential
  setPlayer(prev => {
    if (!prev || !attrKey) return prev;
    const currentVal = prev.attributes[attrKey] as number;
    const newVal = Math.min(prev.potential, currentVal + attrBonus);
    const newXp = (prev.xp ?? 0) + xpBonus;
    return {
      ...prev,
      energy: Math.max(0, (prev.energy ?? 100) - 10),
      xp: newXp,
      attributes: { ...prev.attributes, [attrKey]: newVal }
    };
  });
};
```

```typescript
// Deterministic hash — same name always produces same history
const hashName = (name: string): number =>
  name.split('').reduce((acc, c) => ((acc * 31) + c.charCodeAt(0)) & 0xffff, 0);

const generateClubHistory = (teamName: string, tier: LeagueTier): ClubHistory => {
  const seed = hashName(teamName);
  // Simple LCG — deterministic from seed
  let s = seed;
  const rng = (): number => {
    s = (s * 9301 + 49297) % 233280;
    return s / 233280;
  };

  const [fyMin, fyMax] = tier === LeagueTier.NATIONAL
    ? [1925, 1965] : tier === LeagueTier.STATE
    ? [1965, 1990] : [1980, 2000];

  const foundingYear = fyMin + Math.floor(rng() * (fyMax - fyMin));
  const numPrem = 2 + Math.floor(rng() * 7);
  const premierships = Array.from({ length: numPrem }, () =>
    foundingYear + Math.floor(rng() * (2023 - foundingYear))
  ).sort((a, b) => a - b);

  const [gMin, gMax] = tier === LeagueTier.NATIONAL ? [400, 900] : tier === LeagueTier.STATE ? [200, 500] : [100, 300];
  const [cMin, cMax] = tier === LeagueTier.NATIONAL ? [200, 350] : tier === LeagueTier.STATE ? [100, 200] : [50, 120];

  const RECORD_NAMES = ['J. Smith', 'M. Johnson', 'T. Williams', 'R. Brown', 'D. Wilson', 'S. Murphy', 'C. Taylor'];
  const rName = () => RECORD_NAMES[Math.floor(rng() * RECORD_NAMES.length)];
  const rYear = () => foundingYear + Math.floor(rng() * 60);

  return {
    foundingYear,
    premierships,
    allTimeGoals: { playerName: rName(), value: gMin + Math.floor(rng() * (gMax - gMin)), year: rYear() },
    allTimeCaps:  { playerName: rName(), value: cMin + Math.floor(rng() * (cMax - cMin)), year: rYear() },
    allTimeDisposals: { playerName: rName(), value: (cMin * 15) + Math.floor(rng() * (cMax * 5)), year: rYear() },
  };
};
```
